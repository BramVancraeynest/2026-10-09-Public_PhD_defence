\newcommand\Roadmap[1]{%
\begin{tikzpicture}[
	baseline={(current bounding box.center)},
	station/.style={obj1, draw, circle, fill=UGentBlue, text=white, inner sep=0pt, minimum size={.9*\l*1cm}}
	]
	
	\def\l{.5}
	\def\v{#1}
	\def\d{8.5}    % distance between the stations

    \clip (-.1*\d*\l,-2) rectangle (3.1*\d*\l,3.5);

	\path ({-.2*\l},{-2.9*\l}) rectangle ({(3*\d+.6)*\l},{7.3*\l});
	
	% The road
	\draw[obj1, -{Latex}] (0,0) -- ({3*\d*\l},0);
	
	% Stations: picture above, number on the arrow, message below
	\foreach \k/\fig/\msg in {1/{\DiamondUnitCell}/\partI, 2/{\entangledpair{0}}/\partII, 3/{\PEPS{0}}/\partIII} {
		\ifnum\k>\v \else
			\node at ({(\k-.5)*\d*\l},{4*\l}) {\fig};
			\node[station] at ({(\k-.5)*\d*\l},0) {\k};
			\node[below, text width={3.6cm}, align=flush center] at ({(\k-.5)*\d*\l},{-.6*\l}) {\msg};

	}
