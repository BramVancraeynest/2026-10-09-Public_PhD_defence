% \newpart{n}: announces part n (1, 2 or 3); the other two messages are grey
\newcommand\newpart[1]{%
\begin{frame}{Part #1}
\centering
\def\v{#1}
{\ifnum\v=1 \else\color{gray}\fi \partI}

\bigskip
{\ifnum\v=2 \else\color{gray}\fi \partII}

\bigskip
{\ifnum\v=3 \else\color{gray}\fi \partIII}
\end{frame}}
